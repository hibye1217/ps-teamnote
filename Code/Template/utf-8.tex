Converting \texttt{U+uvwxyz}. ASCII: \texttt{0} to \texttt{7F}; Korean: \texttt{AC00} to \texttt{D7A3}.
\begin{itemize}
	\item \texttt{0} to \texttt{7F}: \texttt{0yyyzzzz}.
	\item \texttt{80} to \texttt{7FF}: \texttt{110xxxyy} \texttt{10yyzzzz}.
	\item \texttt{800} to \texttt{FFFF}: \texttt{1110wwww} \texttt{10xxxxyy} \texttt{10yyzzzz}.
	\item \texttt{10000} to \texttt{10FFFF}: \texttt{11110uvv} \texttt{vvwwww} \texttt{10xxxxyy} \texttt{10yyzzzz}.
\end{itemize}
